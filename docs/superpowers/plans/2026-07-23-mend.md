# Mend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a demoable voice-first post-op ortho recovery co-pilot that fuses spoken check-ins with objective vitals through a deterministic clinical safety engine, for the Juno x Anthropic hackathon (25–26 Jul 2026).

**Architecture:** Deterministic clinical core (recovery graph + fused red-flag engine, fully unit-tested) with LLM edges (Claude for symptom extraction and SBAR generation, ElevenLabs for the voice check-in). Next.js App Router UI on Vercel, Supabase for persistence, a simulated vitals feed standing in for real devices.

**Tech Stack:** Next.js (App Router, TypeScript), Tailwind, Vitest, Zod, @supabase/supabase-js, @anthropic-ai/sdk, ElevenLabs Conversational AI widget. Deploy on Vercel.

## Global Constraints

- Synthetic / anonymised data ONLY. No real patient data anywhere.
- Educational prototype. Every user-facing surface shows a "not medical advice" disclaimer.
- Market = United States. Terminology: ER, 911, care team / surgeon's office, nurse line. No NHS terms.
- The LLM NEVER makes an escalation decision. Only `evaluate()` in the red-flag engine decides green/amber/red.
- Fail-safe direction is ALWAYS toward escalation: on ambiguity or missing/poor vitals, escalate or fall back to symptom-only rules — never reassure on uncertainty.
- Hero procedure = hip arthroplasty (elective THR + hip-fracture hemiarthroplasty). Persona = ~82-year-old post-hip-fracture hemiarthroplasty patient.
- Node 20+, TypeScript strict mode.

```ts
import { VitalsReading } from "./types";

const PLAUSIBLE = {
  hr: [20, 250] as const,
  sbp: [50, 260] as const,
  dbp: [20, 160] as const,
  tempC: [30, 43] as const,
};

function keep(n: number | undefined, range: readonly [number, number]) {
  return n !== undefined && n >= range[0] && n <= range[1] ? n : undefined;
}

export function usableVitals(v: VitalsReading): VitalsReading {
  if (v.quality !== "ok") {
    return { timestamp: v.timestamp, quality: v.quality, ecgFlags: v.ecgFlags };
  }
  return {
    timestamp: v.timestamp,
    quality: v.quality,
    hr: keep(v.hr, PLAUSIBLE.hr),
    sbp: keep(v.sbp, PLAUSIBLE.sbp),
    dbp: keep(v.dbp, PLAUSIBLE.dbp),
    tempC: keep(v.tempC, PLAUSIBLE.tempC),
    ecgFlags: v.ecgFlags,
  };
}
```

```ts
import { Decision, Symptoms, VitalsReading } from "./types";
import { getPhase } from "./recovery-graph";
import { usableVitals } from "./vitals";

export function evaluate(input: { dayPostOp: number; symptoms: Symptoms; vitals: VitalsReading }): Decision {
  const { dayPostOp, symptoms: s } = input;
  const v = usableVitals(input.vitals);
  const phase = getPhase(dayPostOp);
  const r: string[] = [];

  const hr = v.hr;
  const tach = (hr !== undefined && hr > 110) || (v.ecgFlags?.includes("sinus_tachycardia") ?? false);
  const vitalsUsable = v.quality === "ok" && (hr !== undefined || v.sbp !== undefined || v.tempC !== undefined);

  // ---- RED (life-threatening) ----
  if (s.breathless || s.chestPain) {
    if (tach) r.push(`Breathlessness/chest pain with tachycardia (HR ${hr ?? "n/a"}, ECG ${v.ecgFlags?.join("/") ?? "n/a"})`);
    if (tach || !vitalsUsable) {
      return red("Suspected pulmonary embolism", "Call 911 now. This could be a clot on the lung.", "911",
        r.length ? r : ["Breathlessness reported; vitals unavailable — escalating on the safe side"]);
    }
  }
  if (v.sbp !== undefined && v.sbp < 90 && hr !== undefined && hr > 110) {
    return red("Suspected shock / bleeding", "Call 911 now.", "911", [`Hypotension SBP ${v.sbp} with tachycardia HR ${hr}`]);
  }
  if (s.suddenSevereHipPain && (s.legShortenedOrRotated || s.unableToWeightBear)) {
    return red("Suspected hip dislocation", "Go to the ER now — do not put weight on the leg.", "ER",
      ["Sudden severe hip pain with shortened/rotated leg or inability to weight-bear"]);
  }
  if (v.tempC !== undefined && v.tempC >= 38.5 && hr !== undefined && hr > 120) {
    return red("Possible sepsis", "Go to the ER now.", "ER", [`Fever ${v.tempC}C with tachycardia HR ${hr}`]);
  }

  // ---- AMBER (urgent, same-day) ----
  if (s.calfPainOrSwelling) {
    return amber("Possible DVT", "Call your surgeon's office today — you may need a leg scan.", "surgeon_office",
      ["Calf pain/swelling without chest symptoms"]);
  }
  if (s.woundDischarge || (v.tempC !== undefined && v.tempC > phase.normalEnvelope.tempCMax)) {
    return amber("Possible wound infection", "Call your surgeon's office today.", "surgeon_office",
      [s.woundDischarge ? "Wound discharge reported" : `Temp ${v.tempC}C exceeds phase envelope ${phase.normalEnvelope.tempCMax}C`]);
  }
  if (v.ecgFlags?.includes("new_af")) {
    return amber("New atrial fibrillation", "Call the nurse line today.", "nurse_line", ["New AF on ECG, haemodynamically stable"]);
  }
  if (s.painControlled === false) {
    return amber("Uncontrolled pain", "Call the nurse line today to review your pain relief.", "nurse_line", ["Pain not controlled on current analgesia"]);
  }
  if (s.newConfusion) {
    return amber("New confusion", "Call your surgeon's office today.", "surgeon_office", ["New confusion reported"]);
  }

  // ---- GREEN ----
  return { level: "green", action: "You're on track. Here is today's rehab.", rationale: ["No red or amber criteria met"] };
}

function red(condition: string, action: string, call: "911" | "ER", rationale: string[]): Decision {
  return { level: "red", condition, action, call, rationale };
}
function amber(condition: string, action: string, call: "surgeon_office" | "nurse_line", rationale: string[]): Decision {
  return { level: "amber", condition, action, call, rationale };
}
```

```ts
import Anthropic from "@anthropic-ai/sdk";
import { Symptoms } from "@/lib/clinical/types";

const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

const SYMPTOM_TOOL = {
  name: "report_symptoms",
  description: "Extract structured post-op symptoms from the patient's words. Only set a field true if clearly stated. Do NOT infer severity or give advice.",
  input_schema: {
    type: "object" as const,
    properties: {
      breathless: { type: "boolean" }, chestPain: { type: "boolean" },
      calfPainOrSwelling: { type: "boolean" }, woundDischarge: { type: "boolean" },
      feverSubjective: { type: "boolean" }, suddenSevereHipPain: { type: "boolean" },
      legShortenedOrRotated: { type: "boolean" }, unableToWeightBear: { type: "boolean" },
      painControlled: { type: "boolean" }, newConfusion: { type: "boolean" },
    },
  },
};

export async function extractSymptoms(transcript: string): Promise<Symptoms> {
  const res = await client.messages.create({
    model: "claude-sonnet-5",
    max_tokens: 512,
    tools: [SYMPTOM_TOOL],
    tool_choice: { type: "tool", name: "report_symptoms" },
    messages: [{ role: "user", content: `Patient check-in transcript:\n"""${transcript}"""` }],
  });
  const block = res.content.find((b) => b.type === "tool_use");
  return (block && "input" in block ? block.input : {}) as Symptoms;
}
```

```ts
import Anthropic from "@anthropic-ai/sdk";
import { Decision, Symptoms, VitalsReading } from "@/lib/clinical/types";

const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

export async function generateSbar(args: {
  patient: string; dayPostOp: number; procedure: string;
  decision: Decision; symptoms: Symptoms; vitals: VitalsReading;
}): Promise<string> {
  const res = await client.messages.create({
    model: "claude-sonnet-5",
    max_tokens: 600,
    system: "You write concise clinical SBAR handoffs (Situation, Background, Assessment, Recommendation) for a post-op patient to relay to their care team. Use the provided decision and vitals verbatim. Do NOT invent findings. 120 words max.",
    messages: [{ role: "user", content: JSON.stringify(args) }],
  });
  return res.content.map((b) => (b.type === "text" ? b.text : "")).join("");
}
```

```ts
  const symptoms = await extractSymptoms(transcript ?? "");
  const vitals = scenarioVitals(scenario === "pe" ? "pe" : "green", new Date().toISOString());
  const decision = evaluate({ dayPostOp, symptoms, vitals });
  const phase = getPhase(dayPostOp);
  let sbar: string | undefined;
  if (decision.level !== "green") {
    sbar = await generateSbar({ patient: "Margaret (synthetic)", dayPostOp, procedure: "Hip hemiarthroplasty", decision, symptoms, vitals });
  }
  await supabase.from("checkins").insert({ patient_id: null, day_post_op: dayPostOp, symptoms, vitals, decision, sbar });
  return NextResponse.json({ decision, phase, sbar, symptoms, vitals });
}
```
Note: `new Date()` here runs in the Next.js runtime (allowed) — this is app code, not a workflow script.
